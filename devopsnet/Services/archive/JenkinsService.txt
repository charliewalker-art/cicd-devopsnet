using System.Net.Http.Headers;
using System.Net.Http.Json;
using System.Security;
using System.Text;
using devopsnet.Options;
using Microsoft.Extensions.Options;

namespace devopsnet.Services.archive;

public class JenkinsService
{
    private readonly HttpClient _httpClient;
    private readonly JenkinsOptions _options;

    public JenkinsService(HttpClient httpClient, IOptions<JenkinsOptions> options)
    {
        _httpClient = httpClient;
        _options = options.Value;
    }

    public async Task TriggerBuildAsync(string jobName, string repoUrl, string branch, string imageName, string githubToken)
    {
        var queryString = $"REPO_URL={Uri.EscapeDataString(repoUrl)}" +
                           $"&BRANCH={Uri.EscapeDataString(branch)}" +
                           $"&IMAGE_NAME={Uri.EscapeDataString(imageName)}" +
                           $"&GITHUB_TOKEN={Uri.EscapeDataString(githubToken)}";

        var url = $"{_options.BaseUrl}/job/{jobName}/buildWithParameters?{queryString}";

        var request = new HttpRequestMessage(HttpMethod.Post, url);
        request.Headers.Authorization = BuildBasicAuthHeader();

        var response = await _httpClient.SendAsync(request);

        if (!response.IsSuccessStatusCode)
        {
            var error = await response.Content.ReadAsStringAsync();
            throw new InvalidOperationException($"Échec du déclenchement Jenkins : {response.StatusCode} - {error}");
        }
    }

    private AuthenticationHeaderValue BuildBasicAuthHeader()
    {
        var credentials = $"{_options.Username}:{_options.ApiToken}";
        var encodedCredentials = Convert.ToBase64String(Encoding.UTF8.GetBytes(credentials));
        return new AuthenticationHeaderValue("Basic", encodedCredentials);
    }

    public async Task CreateUserAsync(string username, string password, string fullName, string email)
    {
        var groovyScript = $@"
import jenkins.model.Jenkins
import hudson.security.HudsonPrivateSecurityRealm

def realm = Jenkins.get().getSecurityRealm()
def user = realm.createAccount('{EscapeGroovyString(username)}', '{EscapeGroovyString(password)}')
user.setFullName('{EscapeGroovyString(fullName)}')
user.addProperty(new hudson.tasks.Mailer.UserProperty('{EscapeGroovyString(email)}'))
user.save()
Jenkins.get().save()
";

        var request = new HttpRequestMessage(HttpMethod.Post, $"{_options.BaseUrl}/scriptText")
        {
            Content = new FormUrlEncodedContent(new Dictionary<string, string>
            {
                ["script"] = groovyScript,
            }),
        };
        request.Headers.Authorization = BuildBasicAuthHeader();

        var response = await _httpClient.SendAsync(request);
        var responseBody = await response.Content.ReadAsStringAsync();

        if (!response.IsSuccessStatusCode || responseBody.Contains("Exception"))
        {
            throw new InvalidOperationException($"Échec de la création de l'utilisateur Jenkins : {response.StatusCode} - {responseBody}");
        }
    }

    private async Task<JenkinsCrumb?> GetCrumbAsync()
    {
        var request = new HttpRequestMessage(HttpMethod.Get, $"{_options.BaseUrl}/crumbIssuer/api/json");
        request.Headers.Authorization = BuildBasicAuthHeader();

        var response = await _httpClient.SendAsync(request);

        if (!response.IsSuccessStatusCode)
        {
            return null;
        }

        var json = await response.Content.ReadFromJsonAsync<System.Text.Json.JsonElement>();

        return new JenkinsCrumb(
            json.GetProperty("crumbRequestField").GetString()!,
            json.GetProperty("crumb").GetString()!
        );
    }

    private record JenkinsCrumb(string FieldName, string Value);

    private static string EscapeGroovyString(string value)
    {
        return value.Replace("\\", "\\\\").Replace("'", "\\'");
    }

    public async Task CreateOrUpdatePipelineJobAsync(string jobName, string pipelineScript)
    {
        var escapedScript = EscapeGroovyString(pipelineScript);
        var escapedJobName = EscapeGroovyString(jobName);

        var groovyScript = $@"
import jenkins.model.Jenkins
import org.jenkinsci.plugins.workflow.job.WorkflowJob
import org.jenkinsci.plugins.workflow.cps.CpsFlowDefinition

def jenkins = Jenkins.get()
def jobName = '{escapedJobName}'
def existing = jenkins.getItem(jobName)

if (existing != null) {{
    existing.setDefinition(new CpsFlowDefinition('''{escapedScript}''', true))
    existing.save()}} else {{
    def job = jenkins.createProject(WorkflowJob.class, jobName)
    job.setDefinition(new CpsFlowDefinition('''{escapedScript}''', true))
    jenkins.save()}}
";

        var request = new HttpRequestMessage(HttpMethod.Post, $"{_options.BaseUrl}/scriptText")
        {
            Content = new FormUrlEncodedContent(new Dictionary<string, string>
            {
                ["script"] = groovyScript,
            }),
        };
        request.Headers.Authorization = BuildBasicAuthHeader();

        var response = await _httpClient.SendAsync(request);
        var responseBody = await response.Content.ReadAsStringAsync();

        if (!response.IsSuccessStatusCode || responseBody.Contains("Exception"))
        {
            throw new InvalidOperationException($"Échec de la création du job Jenkins : {response.StatusCode} - {responseBody}");
        }
    }

    // AJOUT : Méthode d'automatisation des Credentials Jenkins à la volée
    public async Task CreateOrUpdateGitHubCredentialAsync(string githubToken)
    {
        var credentialXml = $@"<com.cloudbees.plugins.credentials.impl.UsernamePasswordCredentialsImpl>
          <scope>GLOBAL</scope>
          <id>github-user-token</id>
          <description>Token GitHub dynamique injecte par l API .NET</description>
          <username>git</username>
          <password>{SecurityElement.Escape(githubToken)}</password>
          <usernameSecret>false</usernameSecret>
        </com.cloudbees.plugins.credentials.impl.UsernamePasswordCredentialsImpl>";

        var updateUrl = $"{_options.BaseUrl}/credentials/store/system/domain/_/credential/github-user-token/config.xml";
        var updateRequest = new HttpRequestMessage(HttpMethod.Post, updateUrl)
        {
            Content = new StringContent(credentialXml, Encoding.UTF8, "application/xml")
        };
        updateRequest.Headers.Authorization = BuildBasicAuthHeader();

        var updateResponse = await _httpClient.SendAsync(updateRequest);

        if (updateResponse.StatusCode == System.Net.HttpStatusCode.NotFound)
        {
            var createUrl = $"{_options.BaseUrl}/credentials/store/system/domain/_/createCredentials";
            var createRequest = new HttpRequestMessage(HttpMethod.Post, createUrl)
            {
                Content = new StringContent(credentialXml, Encoding.UTF8, "application/xml")
            };
            createRequest.Headers.Authorization = BuildBasicAuthHeader();

            var createResponse = await _httpClient.SendAsync(createRequest);
            if (!createResponse.IsSuccessStatusCode)
            {
                var err = await createResponse.Content.ReadAsStringAsync();
                throw new InvalidOperationException($"Impossible de générer le Credential Jenkins : {err}");
            }
        }
    }

    // MODIFICATION : Utilise la configuration SCM XML pour lire le vrai Jenkinsfile de ton projet privé
    public async Task CreateCloneJobForRepoAsync(string jobName, string repoUrl, string branch)
    {
        var configXml = $@"<?xml version='1.1' encoding='UTF-8'?>
<flow-definition plugin='workflow-job'>
  <actions/>
  <description>Job SCM dynamique cree par l API .NET</description>
  <keepDependencies>false</keepDependencies>
  <properties/>
  <definition class='org.jenkinsci.plugins.workflow.cps.CpsScmFlowDefinition' plugin='workflow-cps'>
    <scm class='hudson.plugins.git.GitSCM' plugin='git'>
      <configVersion>2</configVersion>
      <userRemoteConfigs>
        <hudson.plugins.git.UserRemoteConfig>
          <url>{SecurityElement.Escape(repoUrl)}</url>
          <credentialsId>github-user-token</credentialsId>
        </hudson.plugins.git.UserRemoteConfig>
      </userRemoteConfigs>
      <branches>
        <hudson.plugins.git.BranchSpec>
          <name>*/{SecurityElement.Escape(branch)}</name>
        </hudson.plugins.git.BranchSpec>
      </branches>
      <doGenerateSubmoduleConfigurations>false</doGenerateSubmoduleConfigurations>
      <submoduleCfg class='empty-list'/>
      <extensions/>
    </scm>
    <scriptPath>Jenkinsfile</scriptPath>
    <lightweight>true</lightweight>
  </definition>
  <triggers/>
  <disabled>false</disabled>
</flow-definition>";

        var url = $"{_options.BaseUrl}/job/{jobName}/config.xml";
        var request = new HttpRequestMessage(HttpMethod.Post, url)
        {
            Content = new StringContent(configXml, Encoding.UTF8, "application/xml")
        };
        request.Headers.Authorization = BuildBasicAuthHeader();

        var response = await _httpClient.SendAsync(request);

        if (response.StatusCode == System.Net.HttpStatusCode.NotFound)
        {
            var createUrl = $"{_options.BaseUrl}/createItem?name={Uri.EscapeDataString(jobName)}";
            var createRequest = new HttpRequestMessage(HttpMethod.Post, createUrl)
            {
                Content = new StringContent(configXml, Encoding.UTF8, "application/xml")
            };
            createRequest.Headers.Authorization = BuildBasicAuthHeader();
            var createResponse = await _httpClient.SendAsync(createRequest);
            if (!createResponse.IsSuccessStatusCode)
            {
                throw new InvalidOperationException("Impossible de créer le job Jenkins SCM.");
            }
        }
    }

    // MODIFICATION : Déclenchement propre simplifié pour le mode SCM
    public async Task TriggerCloneJobAsync(string jobName, string githubToken)
    {
        // En mode SCM, le token est géré par l'identifiant global, on utilise un build direct.
        var url = $"{_options.BaseUrl}/job/{jobName}/build";

        var request = new HttpRequestMessage(HttpMethod.Post, url);
        request.Headers.Authorization = BuildBasicAuthHeader();

        var response = await _httpClient.SendAsync(request);

        if (!response.IsSuccessStatusCode)
        {
            var error = await response.Content.ReadAsStringAsync();
            throw new InvalidOperationException($"Échec du déclenchement Jenkins : {response.StatusCode} - {error}");
        }
    }
}
